## Usage

```tsx
import { ReceiptDocument } from "@/components/pdf/receipt";
```

```tsx
<ReceiptDocument
  companyName="Corner Café"
  receiptNumber="004812"
  date="September 13, 2026, 09:42"
  items={[
    { name: "Flat white", qty: 2, total: 9.0 },
    { name: "Almond croissant", qty: 1, total: 4.5 },
    { name: "Sparkling water", qty: 1, total: 3.0 },
  ]}
  subtotal={16.5}
  fees={[{ label: "Tip", value: 2.0 }]}
  total={18.5}
  paymentMethod="card"
  thankYouNote="Thanks for visiting. See you tomorrow."
/>
```
