## Usage

```tsx
import { QuarterlyReportDocument } from "@/components/pdf/blocks/quarterly-report/quarterly-report";
```

```tsx
<QuarterlyReportDocument
  company="Aurora Analytics"
  quarter="Q3 2026"
  subtitle="Performance, customers and priorities for next quarter."
  stats={[
    { label: "Revenue", value: "$1.2M", note: "+18% vs Q2" },
    { label: "Customers", value: "3,480", note: "+412 new" },
    { label: "Churn", value: "1.9%", note: "-0.4 pts" },
  ]}
  trend={{
    title: "Monthly recurring revenue ($k)",
    seriesName: "MRR",
    data: [
      { label: "Apr", value: 310 },
      { label: "May", value: 335 },
      { label: "Jun", value: 362 },
      { label: "Jul", value: 380 },
      { label: "Aug", value: 404 },
      { label: "Sep", value: 431 },
    ],
  }}
  highlights={[
    "Enterprise plan launched in July.",
    "Three new partners in APAC.",
  ]}
  watchList={[
    "Support response time rose to 6h.",
    "Two large renewals due in Q4.",
  ]}
  breakdown={{
    title: "Customers by region",
    data: [
      { label: "North America", value: 1420 },
      { label: "Europe", value: 1180 },
      { label: "APAC", value: 610 },
      { label: "LATAM", value: 270 },
    ],
  }}
  priorities={[
    { label: "Self-serve onboarding", value: 70 },
    { label: "SOC 2 Type II", value: 45 },
    { label: "Mobile app beta", value: 20 },
  ]}
/>
```
